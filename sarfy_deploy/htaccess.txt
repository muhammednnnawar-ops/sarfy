# ========================================================
# HostArmada Speed Reaper - LiteSpeed Web Server Config
# إعدادات متقدمة للحماية القصوى والتشفير والسرعة لتطبيق محفظتي
# ========================================================

# 1. ضبط ترميز اللغة العربية UTF-8 بشكل افتراضي
AddDefaultCharset UTF-8

# 2. أنواع الملفات والـ MIME Types
<IfModule mod_mime.c>
  AddType text/html .html
  AddType text/css .css
  AddType application/javascript .js
  AddType application/json .json
  AddType application/manifest+json .webmanifest
  AddType image/svg+xml .svg
</IfModule>

# 3. تفعيل ضغط Brotli و Gzip عبر LiteSpeed
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/plain text/xml text/css application/javascript application/json image/svg+xml
</IfModule>

# 4. التخزين المؤقت في المتصفح (Browser Caching)
<IfModule mod_expires.c>
  ExpiresActive On
  ExpiresDefault "access plus 1 month"
  ExpiresByType text/html "access plus 1 hour"
  ExpiresByType application/json "access plus 1 day"
  ExpiresByType image/svg+xml "access plus 1 year"
  ExpiresByType application/javascript "access plus 1 month"
  ExpiresByType text/css "access plus 1 month"
</IfModule>

# 5. ترويسات الأمان والحماية المشددة (Military-Grade Security Headers)
<IfModule mod_headers.c>
  # منع استغلال نوع المحتوى
  Header set X-Content-Type-Options "nosniff"
  
  # منع وضع الموقع داخل إطارات خبيثة (Anti-Clickjacking)
  Header set X-Frame-Options "SAMEORIGIN"
  
  # حماية XSS في المتصفحات
  Header set X-XSS-Protection "1; mode=block"
  
  # حماية المراجع
  Header set Referrer-Policy "strict-origin-when-cross-origin"
  
  # تعطيل صلاحيات التتبع والمستشعرات غير الضرورية
  Header set Permissions-Policy "geolocation=(), microphone=(), camera=(), payment=()"
  
  # فرض الاتصال الآمن المشفر HTTPS (HSTS)
  Header set Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" env=HTTPS
  
  # سياسة أمان المحتوى الصارمة (Content Security Policy)
  Header set Content-Security-Policy "default-src 'self' 'unsafe-inline' data: blob: https://fonts.googleapis.com https://fonts.gstatic.com; img-src 'self' data: blob:; font-src 'self' data: https://fonts.gstatic.com;"
</IfModule>

# 6. منع تصفح المجلدات وحماية الملفات الحساسة
Options -Indexes
<FilesMatch "^\.">
  Order allow,deny
  Deny from all
</FilesMatch>
